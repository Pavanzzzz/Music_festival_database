SELECT * 
FROM Books;

SELECT title, price
FROM Books
WHERE price > 300;

SELECT name, city
FROM Customers
WHERE city = 'Mumbai';

SELECT title, stock_quantity
FROM Books
WHERE stock_quantity < 50;

SELECT b.title, a.author_name
FROM Books b
JOIN Authors a
ON b.author_id = a.author_id;

SELECT c.name, o.order_id, o.order_date
FROM Customers c
JOIN Orders o
ON c.customer_id = o.customer_id;

SELECT g.genre_name, COUNT(b.book_id) AS total_books
FROM Genres g
JOIN Books b
ON g.genre_id = b.genre_id
GROUP BY g.genre_name;

SELECT title, price
FROM Books
WHERE price = (
    SELECT MAX(price)
    FROM Books
);

SELECT c.name, SUM(od.quantity_ordered * od.price_at_time_of_order) AS total_spent
FROM Customers c
JOIN Orders o
ON c.customer_id = o.customer_id
JOIN Order_Details od
ON o.order_id = od.order_id
GROUP BY c.name;

SELECT a.author_name, COUNT(b.book_id) AS total_books
FROM Authors a
LEFT JOIN Books b
ON a.author_id = b.author_id
GROUP BY a.author_name
HAVING COUNT(b.book_id) > 1;

SELECT b.title, SUM(od.quantity_ordered) AS total_sold
FROM Books b
JOIN Order_Details od
ON b.book_id = od.book_id
GROUP BY b.title
ORDER BY total_sold DESC;

SELECT c.name, b.title, od.quantity_ordered
FROM Customers c
JOIN Orders o
ON c.customer_id = o.customer_id
JOIN Order_Details od
ON o.order_id = od.order_id
JOIN Books b
ON od.book_id = b.book_id
WHERE od.quantity_ordered >= 1;
